19 // Bilinear resize and BGR-to-RGB conversion helper
20 inline void resize_bgr_to_rgb(const uint8_t *src, int src_w, int src_h, uint8_t *dst, int dst_w, int dst_h)
21 {
22 float x_ratio = ((float)(src_w - 1)) / dst_w;
23 float y_ratio = ((float)(src_h - 1)) / dst_h;
24 for (int i = 0; i < dst_h; i++)
25 {
26 for (int j = 0; j < dst_w; j++)
27 {
28 int x = (int)(x_ratio * j);
29 int y = (int)(y_ratio * i);
30 float x_diff = (x_ratio * j) - x;
31 float y_diff = (y_ratio * i) - y;
32 int src_idx = (y * src_w + x) * 3;
33
34 for (int c = 0; c < 3; c++)
35 {
36 float a = src[src_idx + c];
37 float b = src[src_idx + 3 + c];
38 float d = src[src_idx + src_w * 3 + c];
39 float e = src[src_idx + src_w * 3 + 3 + c];
40
41 float val = a * (1.0f - x_diff) * (1.0f - y_diff) +
42 b * (x_diff) * (1.0f - y_diff) +
43 d * (y_diff) * (1.0f - x_diff) +
44 e * (x_diff) * (y_diff);
45
46 // Write as RGB: destination index c is swapped for R/B (c=0 B -> RGB red; c=2 R -> RGB blue)
47 // BGR to RGB channel reversal
48 dst[(i * dst_w + j) * 3 + (2 - c)] = (uint8_t)val;
49 }
50 }
51 }
52 }

54 inline void crop_and_resize_bgr(
55 const unsigned char *src, int src_w, int src_h,
56 float crop_x, float crop_y, float crop_w, float crop_h,
57 unsigned char *dst, int dst_w, int dst_h)
58 {

60
61 float scale_x = crop_w / static_cast<float>(dst_w);
62 float scale_y = crop_h / static_cast<float>(dst_h);
63
64 for (int y = 0; y < dst_h; ++y)
65 {
66 float src_y = crop_y + (y + 0.5f) * scale_y - 0.5f;
67 src_y = std::max(0.0f, std::min(src_y, static_cast<float>(src_h - 1)));
68 int y0 = static_cast<int>(std::floor(src_y));
69 int y1 = std::min(y0 + 1, src_h - 1);
70 float dy = src_y - y0;
71
72 for (int x = 0; x < dst_w; ++x)
73 {
74 float src_x = crop_x + (x + 0.5f) * scale_x - 0.5f;
75 src_x = std::max(0.0f, std::min(src_x, static_cast<float>(src_w - 1)));
76 int x0 = static_cast<int>(std::floor(src_x));
77 int x1 = std::min(x0 + 1, src_w - 1);
78 float dx = src_x - x0;
79
80 for (int c = 0; c < 3; ++c)
81 {
82 float corners[4] = {
83 static_cast<float>(src[(y0 * src_w + x0) * 3 + c]),
84 static_cast<float>(src[(y0 * src_w + x1) * 3 + c]),
85 static_cast<float>(src[(y1 * src_w + x0) * 3 + c]),
86 static_cast<float>(src[(y1 * src_w + x1) * 3 + c])
87 };
88 float val = (1.0f - dx) * (1.0f - dy) * corners[0] +
89 dx * (1.0f - dy) * corners[1] +
90 (1.0f - dx) * dy * corners[2] +
91 dx * dy * corners[3];
92 dst[(y * dst_w + x) * 3 + c] =
93 static_cast<unsigned char>(std::max(0.0f, std::min(255.0f, val)));
94 }
95 }
96 }

105 inline void crop_and_resize_bgr_with_unroll(
106 const unsigned char *src, int src_w, int src_h,
107 float crop_x, float crop_y, float crop_w, float crop_h,
108 double roll_angle_rad,
109 unsigned char *dst, int dst_w, int dst_h)
110 {
111 if (dst_w <= 0 || dst_h <= 0 || src_w <= 0 || src_h <= 0 || crop_w <= 0.0f || crop_h <= 0.0f) return;
112
113 if (std::abs(roll_angle_rad) < 1e-6)
114 {
115 crop_and_resize_bgr(src, src_w, src_h, crop_x, crop_y, crop_w, crop_h, dst, dst_w, dst_h);
116 return;
117 }
118
119 double cx = src_w * 0.5;
120 double cy = src_h * 0.5;
121 double cos_a = std::cos(roll_angle_rad);
122 double sin_a = std::sin(roll_angle_rad);
123
124 double scale_x = crop_w / static_cast<double>(dst_w);
125 double scale_y = crop_h / static_cast<double>(dst_h);
126
127 double step_x_u = scale_x * cos_a;
128 double step_y_u = -scale_x * sin_a;
129
130 double step_x_v = scale_y * sin_a;
131 double step_y_v = scale_y * cos_a;
132
133 double start_dx = (crop_x + 0.5 * scale_x - 0.5) - cx;
134 double start_dy = (crop_y + 0.5 * scale_y - 0.5) - cy;
135
136 double row_start_x = cx + start_dx * cos_a + start_dy * sin_a;
137 double row_start_y = cy - start_dx * sin_a + start_dy * cos_a;
138
139 for (int y = 0; y < dst_h; ++y)
140 {
141 double cur_src_x = row_start_x;
142 double cur_src_y = row_start_y;
143
144 for (int x = 0; x < dst_w; ++x)
145 {
146 float clamped_x = static_cast<float>(std::max(0.0, std::min(cur_src_x, static_cast<double>(src_w - 1))));
147 float clamped_y = static_cast<float>(std::max(0.0, std::min(cur_src_y, static_cast<double>(src_h - 1))));
148
149 int x0 = static_cast<int>(std::floor(clamped_x));
150 int y0 = static_cast<int>(std::floor(clamped_y));
151 int x1 = std::min(x0 + 1, src_w - 1);
152 int y1 = std::min(y0 + 1, src_h - 1);
153 float dx = clamped_x - static_cast<float>(x0);
154 float dy = clamped_y - static_cast<float>(y0);
155
156 int dst_idx = (y * dst_w + x) * 3;
157
158 for (int c = 0; c < 3; ++c)
159 {
160 float p00 = static_cast<float>(src[(y0 * src_w + x0) * 3 + c]);
161 float p10 = static_cast<float>(src[(y0 * src_w + x1) * 3 + c]);
162 float p01 = static_cast<float>(src[(y1 * src_w + x0) * 3 + c]);
163 float p11 = static_cast<float>(src[(y1 * src_w + x1) * 3 + c]);
164
165 float val = (1.0f - dx) * (1.0f - dy) * p00 +
166 dx * (1.0f - dy) * p10 +
167 (1.0f - dx) * dy * p01 +
168 dx * dy * p11;
169 dst[dst_idx + c] = static_cast<unsigned char>(std::max(0.0f, std::min(255.0f, val)));
170 }
171
172 cur_src_x += step_x_u;
173 cur_src_y += step_y_u;
174 }
175
176 row_start_x += step_x_v;
177 row_start_y += step_y_v;
178 }
179 }

238 // Converts Diagonal FOV to Horizontal FOV given current aspect ratio
239 inline double diagonal_to_horizontal_fov(double diagonal_fov_degrees, double width, double height)
240 {
241 if (width <= 0.0 || height <= 0.0 || diagonal_fov_degrees <= 0.0) return 0.0;
242 double diag_rad = diagonal_fov_degrees * DEG_TO_RAD;
243 double diag_px = std::sqrt(width * width + height * height);
244 double h_fov_rad = 2.0 * std::atan((width / diag_px) * std::tan(diag_rad * 0.5));
245 return h_fov_rad * RAD_TO_DEG;
246 }

281 inline void rotate_image(const unsigned char *src, int w, int h, unsigned char *dst, float angle_rad)
282 {
283 float cos_a = std::cos(angle_rad);
284 float sin_a = std::sin(angle_rad);
285 float cx = w / 2.0f;
286 float cy = h / 2.0f;
287
288 for (int y = 0; y < h; ++y)
289 {
290 float dy = y - cy;
291 for (int x = 0; x < w; ++x)
292 {
293 float dx = x - cx;
294 float src_x = cx + dx * cos_a + dy * sin_a;
295 float src_y = cy - dx * sin_a + dy * cos_a;
296
297 int dst_idx = (y * w + x) * 3;
298
299 // Clamp to valid source coordinates (replicate border) to avoid CNN edge artifacts
300 src_x = std::max(0.0f, std::min(static_cast<float>(w - 1.001f), src_x));
301 src_y = std::max(0.0f, std::min(static_cast<float>(h - 1.001f), src_y));
302
303 int x0 = static_cast<int>(std::floor(src_x));
304 int y0 = static_cast<int>(std::floor(src_y));
305 int x1 = std::min(w - 1, x0 + 1);
306 int y1 = std::min(h - 1, y0 + 1);
307 float tx = src_x - x0;
308 float ty = src_y - y0;
309
310 for (int c = 0; c < 3; ++c)
311 {
312 float p00 = src[(y0 * w + x0) * 3 + c];
313 float p10 = src[(y0 * w + x1) * 3 + c];
314 float p01 = src[(y1 * w + x0) * 3 + c];
315 float p11 = src[(y1 * w + x1) * 3 + c];
316
317 float val = (1.0f - tx) * (1.0f - ty) * p00 +
318 tx * (1.0f - ty) * p10 +
319 (1.0f - tx) * ty * p01 +
320 tx * ty * p11;
321 dst[dst_idx + c] = static_cast<unsigned char>(std::max(0.0f, std::min(255.0f, val)));
322 }
323 }
324 }
325 }
