19 // Bilinear resize and BGR-to-RGB conversion helper
20 inline void resize_bgr_to_rgb(const uint8_t *src, int src_w, int src_h, uint8_t *dst, int dst_w, int dst_h)
21 {
22 float x_ratio = ((float)(src_w - 1)) / dst_w;
23 float y_ratio = ((float)(src_h - 1)) / dst_h;
24 for (int i = 0; i < dst_h; i++)
25 {
26 for (int j = 0; j < dst_w; j++)
27 {
28 int x = (int)(x_ratio * j);
29 int y = (int)(y_ratio * i);
30 float x_diff = (x_ratio * j) - x;
31 float y_diff = (y_ratio * i) - y;
32 int src_idx = (y * src_w + x) * 3;
33
34 for (int c = 0; c < 3; c++)
35 {
36 float a = src[src_idx + c];
37 float b = src[src_idx + 3 + c];
38 float d = src[src_idx + src_w * 3 + c];
39 float e = src[src_idx + src_w * 3 + 3 + c];
40
41 float val = a * (1.0f - x_diff) * (1.0f - y_diff) +
42 b * (x_diff) * (1.0f - y_diff) +
43 d * (y_diff) * (1.0f - x_diff) +
44 e * (x_diff) * (y_diff);
45
46 // Write as RGB: destination index c is swapped for R/B (c=0 B -> RGB red; c=2 R -> RGB blue)
47 // BGR to RGB channel reversal
48 dst[(i * dst_w + j) * 3 + (2 - c)] = (uint8_t)val;
49 }
50 }
51 }
52 }

54 inline void crop_and_resize_bgr(
55 const unsigned char *src, int src_w, int src_h,
56 float crop_x, float crop_y, float crop_w, float crop_h,
57 unsigned char *dst, int dst_w, int dst_h)
58 {
59 if (dst_w <= 0 || dst_h <= 0 || src_w <= 0 || src_h <= 0 || crop_w <= 0.0f || crop_h <= 0.0f) return;
60
61 float scale_x = crop_w / static_cast<float>(dst_w);
62 float scale_y = crop_h / static_cast<float>(dst_h);
63
64 for (int y = 0; y < dst_h; ++y)
65 {
66 float src_y = crop_y + (y + 0.5f) * scale_y - 0.5f;
67 src_y = std::max(0.0f, std::min(src_y, static_cast<float>(src_h - 1)));
68 int y0 = static_cast<int>(std::floor(src_y));
69 int y1 = std::min(y0 + 1, src_h - 1);
70 float dy = src_y - y0;
71
72 for (int x = 0; x < dst_w; ++x)
73 {
74 float src_x = crop_x + (x + 0.5f) * scale_x - 0.5f;
75 src_x = std::max(0.0f, std::min(src_x, static_cast<float>(src_w - 1)));
76 int x0 = static_cast<int>(std::floor(src_x));
77 int x1 = std::min(x0 + 1, src_w - 1);
78 float dx = src_x - x0;
79
80 for (int c = 0; c < 3; ++c)
81 {
82 float corners[4] = {
83 static_cast<float>(src[(y0 * src_w + x0) * 3 + c]),
84 static_cast<float>(src[(y0 * src_w + x1) * 3 + c]),
85 static_cast<float>(src[(y1 * src_w + x0) * 3 + c]),
86 static_cast<float>(src[(y1 * src_w + x1) * 3 + c])
87 };
88 float val = (1.0f - dx) * (1.0f - dy) * corners[0] +
89 dx * (1.0f - dy) * corners[1] +
90 (1.0f - dx) * dy * corners[2] +
91 dx * dy * corners[3];
92 dst[(y * dst_w + x) * 3 + c] =
93 static_cast<unsigned char>(std::max(0.0f, std::min(255.0f, val)));
94 }
95 }
96 }

146 // Computes the expected pixel width of a credit card on screen for a given HFOV and distance
147 inline double get_card_width_px(double fov_degrees, double card_distance_mm, double frame_width, double card_width_mm = 85.603)
148 {
149 if (card_distance_mm <= 0.0) return 0.0;

156 // Converts Diagonal FOV to Horizontal FOV given current aspect ratio
157 inline double diagonal_to_horizontal_fov(double diagonal_fov_degrees, double width, double height)
158 {
159 if (width <= 0.0 || height <= 0.0 || diagonal_fov_degrees <= 0.0) return 0.0;
160 double diag_rad = diagonal_fov_degrees * DEG_TO_RAD;
161 double diag_px = std::sqrt(width * width + height * height);
162 double h_fov_rad = 2.0 * std::atan((width / diag_px) * std::tan(diag_rad * 0.5));
163 return h_fov_rad * RAD_TO_DEG;
164 }

199 inline void rotate_image(const unsigned char *src, int w, int h, unsigned char *dst, float angle_rad)
200 {
201 float cos_a = std::cos(angle_rad);
202 float sin_a = std::sin(angle_rad);
203 float cx = w / 2.0f;
204 float cy = h / 2.0f;
205
206 for (int y = 0; y < h; ++y)
207 {
208 float dy = y - cy;
209 for (int x = 0; x < w; ++x)
210 {
211 float dx = x - cx;
212 float src_x = cx + dx * cos_a + dy * sin_a;
213 float src_y = cy - dx * sin_a + dy * cos_a;
214
215 int dst_idx = (y * w + x) * 3;
216
217 // Clamp to valid source coordinates (replicate border) to avoid CNN edge artifacts
218 src_x = std::max(0.0f, std::min(static_cast<float>(w - 1.001f), src_x));
219 src_y = std::max(0.0f, std::min(static_cast<float>(h - 1.001f), src_y));
220
221 int x0 = static_cast<int>(std::floor(src_x));
222 int y0 = static_cast<int>(std::floor(src_y));
223 int x1 = std::min(w - 1, x0 + 1);
224 int y1 = std::min(h - 1, y0 + 1);
225 float tx = src_x - x0;
226 float ty = src_y - y0;
227
228 for (int c = 0; c < 3; ++c)
229 {
230 float p00 = src[(y0 * w + x0) * 3 + c];
231 float p10 = src[(y0 * w + x1) * 3 + c];
232 float p01 = src[(y1 * w + x0) * 3 + c];
233 float p11 = src[(y1 * w + x1) * 3 + c];
234
235 float val = (1.0f - tx) * (1.0f - ty) * p00 +
236 tx * (1.0f - ty) * p10 +
237 (1.0f - tx) * ty * p01 +
238 tx * ty * p11;
239 dst[dst_idx + c] = static_cast<unsigned char>(std::max(0.0f, std::min(255.0f, val)));
240 }
241 }
242 }
243 }
