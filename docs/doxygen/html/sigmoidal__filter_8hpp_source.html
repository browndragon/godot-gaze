70 float update(bool has_sample, float sample_val, float confidence = 1.0f, double timestamp_s = -1.0) {
71 double dt = (last_timestamp_s >= 0.0 && timestamp_s >= last_timestamp_s)
72 ? (timestamp_s - last_timestamp_s)
73 : (1.0 / 60.0);
74 last_timestamp_s = timestamp_s;
75 if (dt > 0.5) dt = 0.5;
76
77 if (has_sample) {
78 time_lost_s = 0.0;
79 float c = std::clamp(confidence, 0.1f, 1.0f);
80 float tau = base_tau_track_s / c;
81 float alpha = static_cast<float>(1.0 - std::exp(-dt / tau));
82 current_val += alpha * (sample_val - current_val);
83 last_valid_val = current_val;
84 } else {
85 time_lost_s += dt;
86 float t_norm = std::clamp((static_cast<float>(time_lost_s) - hold_window_s) / (decay_window_s - hold_window_s), 0.0f, 1.0f);
87 // Smoothstep Hermite S-curve: S(0) = 0, S(1) = 1, S'(0) = 0, S'(1) = 0
88 float decay_weight = 1.0f - (t_norm * t_norm * (3.0f - 2.0f * t_norm));
89 current_val = neutral_val + (last_valid_val - neutral_val) * decay_weight;
90 if (std::abs(current_val - neutral_val) < min_cutoff_delta || t_norm >= 1.0f) {
91 current_val = neutral_val;
92 }
93 }
94 return current_val;
95 }
